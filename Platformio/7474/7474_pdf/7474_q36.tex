\subsection{Breadboard}

The breadboard is used for temporary circuit connections.

\subsection{IC 7474}

The 7474 is a dual D Flip-Flop IC used to store binary data.

\subsection{IC 7447}

The 7447 is a BCD to Seven Segment Decoder Driver.

\subsection{Arduino Uno}

Arduino Uno is used to generate the clock signal and read the outputs from the 7474 ICs.

\section{Software Implementation}

\begin{verbatim}
#include <Arduino.h>

int Z;
int D,C,B,A;

void disp_7447(int D,int C,int B,int A)
{
digitalWrite(2,A);
digitalWrite(3,B);
digitalWrite(4,C);
digitalWrite(5,D);
}

void setup()
{
pinMode(2,OUTPUT);
pinMode(3,OUTPUT);
pinMode(4,OUTPUT);
pinMode(5,OUTPUT);

pinMode(6,INPUT);
pinMode(7,INPUT);
pinMode(8,INPUT);
pinMode(9,INPUT);

pinMode(13,OUTPUT);
}

void loop()
{
digitalWrite(13,HIGH);
delay(10);
digitalWrite(13,LOW);

Z = digitalRead(9);

int F = !Z;

if(F==0)
{
D=0; C=0; B=0; A=0;
}
else
{
D=0; C=0; B=0; A=1;
}

disp_7447(D,C,B,A);
delay(1000);
}
\end{verbatim}
